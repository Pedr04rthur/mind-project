// ---------- Erro ----------

export class ApiError extends Error {
  status: number;
  code?: string;

  constructor(message: string, status: number, code?: string) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.code = code;
  }
}

interface RequestOptions extends RequestInit {
  skipRefresh?: boolean;
}

// ---------- Fetch core ----------

async function rawRequest(
  path: string,
  init?: RequestOptions
): Promise<Response> {
  const headers = new Headers(init?.headers);

  if (init?.body && !headers.has("Content-Type")) {
    headers.set("Content-Type", "application/json");
  }

  if (accessToken) {
    headers.set("Authorization", `Bearer ${accessToken}`);
  }

  return fetch(`${BASE}${path}`, {
    ...init,
    headers,
    credentials: "include",
  });
}

async function parseError(res: Response): Promise<ApiError> {
  let message = `Erro ${res.status}`;
  let code: string | undefined;

  try {
    const body = await res.json();

    if (body?.message) message = body.message;
    if (body?.code) code = body.code;

    if (!body?.message) {
      if (res.status === 400)
        message = "Dados inválidos.";
      else if (res.status === 401)
        message = "Sessão expirada. Faça login novamente.";
      else if (res.status === 403)
        message = "Sem permissão para esta ação.";
      else if (res.status === 404)
        message = "Recurso não encontrado.";
      else if (res.status === 409)
        message = "Conflito: recurso já existe.";
      else if (res.status >= 500)
        message = "Erro no servidor.";
    }
  } catch {
    if (res.status === 401)
      message = "Sessão expirada. Faça login novamente.";
    else if (res.status >= 500)
      message = "Erro no servidor.";
  }

  return new ApiError(message, res.status, code);
}

async function tryRefresh(): Promise<boolean> {
  try {
    const res = await fetch(`${BASE}/auth/refresh`, {
      method: "POST",
      credentials: "include",
    });

    if (!res.ok) return false;

    const data = (await res.json()) as { token: string };

    accessToken = data.token;

    return true;
  } catch {
    return false;
  }
}

async function request<T>(
  path: string,
  init?: RequestOptions
): Promise<T> {
  let res = await rawRequest(path, init);

  if (res.status === 401 && !init?.skipRefresh) {
    const ok = await tryRefresh();

    if (ok) {
      res = await rawRequest(path, init);
    } else {
      accessToken = null;
      onUnauthorized?.();
    }
  }

  if (!res.ok) throw await parseError(res);

  if (res.status === 204) return undefined as T;

  const text = await res.text();

  return text
    ? (JSON.parse(text) as T)
    : (undefined as T);
}

export function cpfToApi(cpf: string): string {
  return cpf.replace(/\D/g, "");
}

export function cpfFromApi(cpf: string): string {
  const digits = cpf.replace(/\D/g, "").padStart(11, "0").slice(0, 11);
  return digits.replace(/(\d{3})(\d{3})(\d{3})(\d{2})/, "$1.$2.$3-$4");
}

export function patientFromApi(api: PatientApi): Paciente {
  return {
    id: api.id,
    cpf: cpfFromApi(api.cpf),
    nome: api.name,
    telefone: api.phone,
    endereco: api.address,
    email: api.email,
    status: "ATIVO",
    prioridade: "BAIXA",
  };
}

    name: paciente.nome,

    password: senha,
  };
}

/**
 * Data local no formato YYYY-MM-DD (usa o fuso do navegador, não UTC).
 */
export function getDataDiaAtual(date: Date = new Date()): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

/**
 * Hora no formato HH:mm.
 */
export function formatarHora(iso: string): string {
  const d = new Date(iso);
  const h = String(d.getHours()).padStart(2, "0");
  const m = String(d.getMinutes()).padStart(2, "0");
  return `${h}:${m}`;
}

/**
 * Formato que o backend /schedule espera: DD-MM-YYYY
 */
export function toBackendDate(isoDate: string): string {
  const [year, month, day] = isoDate.split("-");
  return `${day}-${month}-${year}`;
}

const STORAGE_KEY = "mindcare:usuario";

interface AuthContextValue {
  usuario: UsuarioLogado | null;
  isAutenticado: boolean;
  entrar: (email: string, senha: string) => Promise<PerfilUsuario>;
  sair: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | null>(null);

function loadInitial(): UsuarioLogado | null {
  try {
    const raw = sessionStorage.getItem(STORAGE_KEY);
    if (raw) return JSON.parse(raw) as UsuarioLogado;
  } catch {
    // ignora
  }
  return null;
}

export function AuthProvider({ children }: { children: ReactNode }) {
  const [usuario, setUsuario] = useState<UsuarioLogado | null>(loadInitial);

  useEffect(() => {
    try {
      if (usuario) sessionStorage.setItem(STORAGE_KEY, JSON.stringify(usuario));
      else sessionStorage.removeItem(STORAGE_KEY);
    } catch {
      // ignora
    }
  }, [usuario]);

  useEffect(() => {
    setOnUnauthorized(() => {
      setUsuario(null);
      setAccessToken(null);
    });
    return () => setOnUnauthorized(null);
  }, []);

  const entrar = useCallback(async (email: string, senha: string) => {
    const auth = await api.auth.signin(email, senha);
    setAccessToken(auth.token);

    const me = await api.auth.me();
    const novo = usuarioFromMe(me);

    setUsuario(novo);
    return novo.perfil;
  }, []);

  const sair = useCallback(async () => {
    try {
      await api.auth.logout();
    } catch {
      // ignora
    } finally {
      setAccessToken(null);
      setUsuario(null);
    }
  }, []);

  return (
    <AuthContext.Provider
      value={{ usuario, isAutenticado: usuario !== null, entrar, sair }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) {
    throw new Error("useAuth precisa estar dentro de <AuthProvider>");
  }
  return ctx;
}

interface PacientesContextValue {
  pacientes: Paciente[];
  loading: boolean;
  erro: string | null;
  getByCpf: (cpf: string) => Paciente | undefined;
  create: (paciente: Paciente, senha: string) => Promise<void>;
  update: (paciente: Paciente, senhaPlaceholder?: string) => Promise<void>;
  remove: (cpf: string) => Promise<void>;
  refresh: () => Promise<void>;
}

const PacientesContext =
  createContext<PacientesContextValue | null>(null);

export function PacientesProvider({
  children,
}: {
  children: ReactNode;
}) {
  const { usuario } = useAuth();

  const perfil = usuario?.perfil;
  const email = usuario?.emailLogin;

  const [pacientes, setPacientes] = useState<Paciente[]>([]);
  const [loading, setLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const refresh = useCallback(async () => {
    if (!email || !perfil) {
      setPacientes([]);
      setLoading(false);
      setErro(null);
      return;
    }

    setLoading(true);
    setErro(null);

    try {
      if (perfil === "RECEPCAO") {
        const page = await api.patients.list(0, 200);

        setPacientes(
          page.content.map(patientFromApi)
        );

        return;
      }

      if (perfil === "PROFISSIONAL") {
        const consultas = await api.appointments.listMine();

        const pacientesUnicos = new Map<number, Paciente>();

        for (const consulta of consultas) {
          if (consulta.status === "CANCELLED") {
            continue;
          }

          const paciente = consulta.patient;

          if (pacientesUnicos.has(paciente.id)) {
            continue;
          }

          pacientesUnicos.set(paciente.id, {
            id: paciente.id,
            cpf: cpfFromApi(paciente.cpf),
            nome: paciente.name,
            telefone: "",
            endereco: "",
            email: "",
          });
        }

        setPacientes(
          Array.from(pacientesUnicos.values()).sort(
            (a, b) => a.nome.localeCompare(b.nome, "pt-BR")
          )
        );

        return;
      }

      setPacientes([]);
    } catch (err) {
      const msg =
        err instanceof ApiError
          ? err.message
          : err instanceof Error
          ? err.message
          : "Erro ao carregar pacientes.";

      setPacientes([]);
      setErro(msg);

      console.error(
        "[MindCare] Erro ao carregar pacientes:",
        err
      );
    } finally {
      setLoading(false);
    }
  }, [email, perfil]);

  useEffect(() => {
    setPacientes([]);
    void refresh();
  }, [refresh]);

  useEffect(() => {
    if (perfil !== "PROFISSIONAL") return;

    const atualizar = () => {
      void refresh();
    };

    window.addEventListener("focus", atualizar);

    const intervalo = window.setInterval(
      atualizar,
      30000
    );

    return () => {
      window.removeEventListener("focus", atualizar);
      window.clearInterval(intervalo);
    };
  }, [perfil, refresh]);

  const getByCpf = (cpf: string) => {
    const alvo = cpfFromApi(cpf);

    return pacientes.find(
      (p) => p.cpf === alvo
    );
  };

  const create = async (
    paciente: Paciente,
    senha: string
  ) => {
    await api.patients.create(
      patientToApi(paciente, senha)
    );

    await refresh();
  };

  const update = async (
    paciente: Paciente,
    senhaPlaceholder = "senha123"
  ) => {
    await api.patients.update(
      cpfToApi(paciente.cpf),
      patientToApi(paciente, senhaPlaceholder)
    );

    await refresh();
  };

  const remove = async (cpf: string) => {
    await api.patients.remove(cpfToApi(cpf));
    await refresh();
  };

  return (
    <PacientesContext.Provider
      value={{
        pacientes,
        loading,
        erro,
        getByCpf,
        create,
        update,
        remove,
        refresh,
      }}
    >
      {children}
    </PacientesContext.Provider>
  );
}

export function usePacientes() {
  const ctx = useContext(PacientesContext);

  if (!ctx) {
    throw new Error(
      "usePacientes precisa estar dentro de <PacientesProvider>"
    );
  }

  return ctx;
}

  const [mostrarSenha, setMostrarSenha] = useState(false);
  const [loading, setLoading] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  const [perfilSelecionado, setPerfilSelecionado] =
    useState<PerfilUsuario | null>(null);
  const [perfilAberto, setPerfilAberto] = useState(false);

  const opcaoAtual = PERFIS.find((p) => p.id === perfilSelecionado) ?? null;

  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    setErro(null);

    if (!perfilSelecionado) {
      setErro("Selecione seu perfil para continuar.");
      return;
    }
    if (!login.trim() || !senha) {
      setErro("Preencha login e senha para continuar.");
      return;
    }

    setLoading(true);
    try {
      const perfilReal = await entrar(login.trim(), senha);

      if (perfilReal !== perfilSelecionado) {
        setErro(
          `Essa conta é de ${ROTULO_PERFIL[perfilReal]}, não de ${ROTULO_PERFIL[perfilSelecionado]}.`
        );
        return;
      }

      navigate(getHomeDoPerfil(perfilReal), { replace: true });
    } catch (err) {
      if (err instanceof ApiError) {
        if (err.status === 401) setErro("Email ou senha inválidos.");
        else setErro(err.message);
      } else {
        setErro("Não foi possível entrar. Tente novamente.");
      }
    } finally {
      setLoading(false);
    }
  };

  return (
    <main className={styles.page}>
      <section className={styles.brandPanel}>
        <div className={styles.brandContent}>
          <div className={styles.brandMark}>
            <img src={mindguy} alt="MindCare" className={styles.brandLogo} />
            <span>MindCare</span>
          </div>
          <h1 className={styles.brandTitle}>
            Cuidado contínuo em saúde mental.
          </h1>
          <p className={styles.brandSubtitle}>
            Acompanhe seu bem-estar, compartilhe seu diário com seu terapeuta
            e mantenha sua jornada terapêutica em um só lugar.
          </p>
        </div>
      </section>

      <section className={styles.formPanel}>
        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <header className={styles.formHeader}>
            <div className={styles.mobileBrand}>
              <img
                src={mindguy}
                alt="MindCare"
                className={styles.mobileBrandLogo}
              />
              <span>MindCare</span>
            </div>
            <h2 className={styles.formTitle}>Bem-vindo</h2>
            <p className={styles.formSubtitle}>Faça o login</p>
          </header>

          <div className={styles.perfilField}>
            <label className={styles.label}>Selecione seu perfil</label>

            <button
              type="button"
              className={`${styles.perfilTrigger} ${
                perfilAberto ? styles.perfilTriggerAberto : ""
              }`}
              onClick={() => setPerfilAberto((v) => !v)}
              aria-expanded={perfilAberto}
              aria-haspopup="listbox"
            >
              {opcaoAtual ? (
                <span className={styles.perfilTriggerConteudo}>
                  <span className={styles.perfilTriggerIcone}>
                    <opcaoAtual.icon size={18} strokeWidth={2} />

                  <span className={styles.perfilTriggerTextos}>

                    <span className={styles.perfilTriggerHint}>

              ) : (

              <ChevronDown
                size={18}

                }`}

                      type="button"
                      role="option"
                      aria-selected={ativo}
                      className={`${styles.perfilOpcao} ${
                        ativo ? styles.perfilOpcaoAtiva : ""

                        setPerfilAberto(false);

                    >
                      <span className={styles.perfilOpcaoIcone}>
                        <Icone size={18} strokeWidth={2} />
                      </span>
                      <span className={styles.perfilOpcaoTextos}>
                        <span className={styles.perfilOpcaoLabel}>
                          {opcao.label}
                        </span>
                        <span className={styles.perfilOpcaoHint}>
                          {opcao.hint}
                        </span>
                      </span>
                    </button>
                  );
                })}
              </div>
            )}
          </div>

          <div className={styles.fieldGroup}>
            <label htmlFor="login" className={styles.label}>
              Email ou CPF
            </label>
            <div className={styles.inputWrapper}>
              <User size={18} className={styles.inputIcon} />
              <input
                id="login"
                type="text"
                autoComplete="username"
                className={styles.input}
                placeholder="voce@exemplo.com"

                onChange={(e) => setLogin(e.target.value)}

import { Navigate, Outlet } from "react-router-dom";
import { useAuth, getHomeDoPerfil } from "../../contexts/AuthContext";
import type { PerfilUsuario } from "../../types/auth";

interface RequirePerfilProps {
  permitidos: PerfilUsuario[];
}

export function RequirePerfil({ permitidos }: RequirePerfilProps) {
  const { usuario } = useAuth();

  if (!usuario) {
    return <Navigate to="/login" replace />;
  }

  if (!permitidos.includes(usuario.perfil)) {
    return <Navigate to={getHomeDoPerfil(usuario.perfil)} replace />;
  }

  return <Outlet />;
}
import { useEffect, useState } from "react";

import {
  ShieldCheck,
  Search,
  RefreshCw,
  Database,
  AlertCircle,
  ChevronLeft,
  ChevronRight,
  FileSearch,
  Filter,
  X,
  Lock,
} from "lucide-react";

import { auditoriaApi } from "../../../services/api";

import type {
  FiltrosAuditoria,
  PaginaAuditoria,
} from "../../../types/auditoria";

import styles from "./AuditoriaPage.module.css";

const FILTROS_INICIAIS: FiltrosAuditoria = {
  cpf: "",
  tabela: "",
  inicio: "",
  fim: "",

  size: 10,
};

const TABELAS: Record<string, string> = {
  patients: "Pacientes",
  professionals: "Profissionais",
  moods: "Registros de humor",
  consultations: "Consultas",
  prescriptions: "Prescrições",
};

function formatarCPF(cpf: string | null) {
  if (!cpf) return "—";

  return cpf.replace(
    /(\d{3})(\d{3})(\d{3})(\d{2})/,
    "$1.$2.$3-$4"
  );
}

function formatarData(data: string) {
  return new Date(data).toLocaleString("pt-BR");
}

export function AuditoriaPage() {

    useState<FiltrosAuditoria>(FILTROS_INICIAIS);

  const [filtros, setFiltros] =
    useState<FiltrosAuditoria>(FILTROS_INICIAIS);

  const [dados, setDados] =
    useState<PaginaAuditoria | null>(null);

  const [loading, setLoading] = useState(true);

  const [erro, setErro] =
    useState<string | null>(null);

  const [versao, setVersao] = useState(0);

  useEffect(() => {
    let ativo = true;

    async function carregar() {
      setLoading(true);
      setErro(null);
      setDados(null);

      try {
        const resultado =
          await auditoriaApi.listar(filtros);

        if (ativo) setDados(resultado);

      } catch (err) {
        if (!ativo) return;

        setErro(
          err instanceof Error
            ? err.message
            : "Não foi possível consultar a auditoria."
        );

      } finally {
        if (ativo) setLoading(false);
      }
    }

    return () => {
      ativo = false;
    };
  }, [filtros, versao]);

  function pesquisar() {
    if (
      form.inicio &&
      form.fim &&
      form.inicio > form.fim
    ) {
      setErro("A data inicial deve ser anterior à final.");
      return;
    }

    setFiltros({
      ...form,
      cpf: form.cpf?.replace(/\D/g, ""),
      page: 0,
    });
  }

  function limpar() {
    setForm({ ...FILTROS_INICIAIS });
    setFiltros({ ...FILTROS_INICIAIS });

  function mudarPagina(page: number) {
    setFiltros((anterior) => ({
      ...anterior,
      page,
    }));
  }

  const registros = dados?.content ?? [];
  const pagina = dados?.number ?? 0;
  const totalPaginas = dados?.totalPages ?? 0;

  return (

        <div className={styles.heading}>
          <div className={styles.titleIcon}>
            <ShieldCheck size={27} />
          </div>

          <div>
            <h1>Auditoria LGPD</h1>
            <p>
              Rastreabilidade das exclusões de dados do MindCare.
            </p>
          </div>
        </div>

        <button
          className={styles.secondaryButton}
          onClick={() => setVersao((v) => v + 1)}
          disabled={loading}
        >
          <RefreshCw size={16} />
          Atualizar
        </button>
      </header>

      <div className={styles.stats}>

        <article className={styles.stat}>
          <div className={styles.statIcon}>
            <Database size={22} />
          </div>

          <div>
            <span>Exclusões encontradas</span>
            <strong>{dados?.totalElements ?? "—"}</strong>
          </div>
        </article>

        <article className={styles.stat}>
          <div className={styles.statIcon}>
            <FileSearch size={22} />
          </div>

          <div>
            <span>Registros nesta página</span>
            <strong>{dados ? registros.length : "—"}</strong>
          </div>
        </article>

        <article className={styles.stat}>
          <div className={styles.statIcon}>
            <Lock size={22} />
          </div>

          <div>
            <span>Permissão de acesso</span>
            <strong className={styles.smallValue}>
              Recepção
            </strong>
          </div>
        </article>

      </div>

      <section className={styles.notice}>
        <ShieldCheck size={22} />

        <div>
          <strong>Registro de exclusões</strong>
          <p>
            Os dados excluídos são registrados no backend
            em formato JSON. Esta tela apresenta os
            metadados necessários para rastreabilidade,
            sem expor o conteúdo sensível dos snapshots.
          </p>
        </div>
      </section>

      <section className={styles.panel}>

        <h2>
          <Filter size={18} />
          Filtros de auditoria
        </h2>

        <div className={styles.filters}>

          <label>
            CPF afetado

            <input
              placeholder="000.000.000-00"
              value={form.cpf ?? ""}
              maxLength={14}
              onChange={(e) =>
                setForm({
                  ...form,
                  cpf: e.target.value,
                })
              }
            />
          </label>

          <label>
            Entidade

            <select
              value={form.tabela ?? ""}
              onChange={(e) =>
                setForm({
                  ...form,
                  tabela: e.target.value,
                })
              }
            >
              <option value="">Todas</option>
              <option value="patients">Pacientes</option>
              <option value="professionals">
                Profissionais
              </option>
            </select>
          </label>

          <label>
            Data inicial

            <input
              type="date"
              value={form.inicio ?? ""}
              onChange={(e) =>
                setForm({
                  ...form,
                  inicio: e.target.value,
                })
              }
            />
          </label>

          <label>
            Data final

            <input
              type="date"
              value={form.fim ?? ""}
              onChange={(e) =>
                setForm({
                  ...form,
                  fim: e.target.value,
                })
              }
            />
          </label>

        </div>

        <div className={styles.actions}>
          <button
            className={styles.secondaryButton}
            onClick={limpar}
          >
            <X size={16} />
            Limpar
          </button>

          <button
            className={styles.primaryButton}
            onClick={pesquisar}
            disabled={loading}
          >
            <Search size={16} />
            Pesquisar
          </button>
        </div>

      </section>

      {erro && (
        <div className={styles.error}>
          <AlertCircle size={19} />

      )}

      <section className={styles.panel}>

          <h2>Histórico de exclusões</h2>

        </div>

          </div>
        ) : (
          <div className={styles.tableWrapper}>
            <table className={styles.table}>

              <thead>
                <tr>
                  <th>ID</th>
                  <th>Entidade</th>
                  <th>Registro</th>
                  <th>CPF afetado</th>
                  <th>Responsável</th>
                  <th>Data e horário</th>
                  <th>Operação</th>
                </tr>
              </thead>

              <tbody>
                {registros.map((log) => (
                  <tr key={log.id}>

                    <td>#{log.id}</td>

                    <td>
                      <strong>
                        {TABELAS[log.tabelaAfetada]

                      </strong>
                    </td>

                    <td>#{log.registroId}</td>

                    <td>{formatarCPF(log.cpfAfetado)}</td>

                    <td>{log.responsavel}</td>

                    <td>{formatarData(log.dataExclusao)}</td>

                    <td>
                      <span className={styles.badge}>
                        Exclusão
                      </span>
                    </td>

                  </tr>
                ))}
              </tbody>

            </table>
          </div>
        )}

        <footer className={styles.pagination}>

          <span>
            Página {totalPaginas ? pagina + 1 : 0}
            {" de "}
            {totalPaginas}
          </span>

          <div>
            <button
              className={styles.secondaryButton}
              disabled={loading || pagina === 0}
              onClick={() => mudarPagina(pagina - 1)}
            >
              <ChevronLeft size={16} />
              Anterior
            </button>

            <button
              className={styles.secondaryButton}
              disabled={
                loading || pagina + 1 >= totalPaginas
              }
              onClick={() => mudarPagina(pagina + 1)}
            >
              Próxima
              <ChevronRight size={16} />
            </button>
          </div>

        </footer>

      </section>

    </main>
